class Solution {
public:
    vector<string> createGrid(int m, int n) {
        vector<string> result(m, string(n, '#'));

        for(int i = 0; i < m; i++) {
           result[i][0] = '.';
        }

        for(int j = 0; j < n; j++) {
            result[m - 1][j] = '.';
        }
        
        return result;
    }
};
