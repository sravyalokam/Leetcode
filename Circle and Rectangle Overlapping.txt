/**
 * @param {number} radius
 * @param {number} xCenter
 * @param {number} yCenter
 * @param {number} x1
 * @param {number} y1
 * @param {number} x2
 * @param {number} y2
 * @return {boolean}
 */
var checkOverlap = function(radius, xCenter, yCenter, x1, y1, x2, y2) {
    const circleMinX = xCenter - radius;
    const circleMaxX = xCenter + radius;
    const circleMinY = yCenter - radius;
    const circleMaxY = yCenter + radius;

    const rectMinX = x1;
    const rectMaxX = x2;
    const rectMinY = y1;
    const rectMaxY = y2;

    return (
        Math.max(circleMinX, rectMinX) <= Math.min(circleMaxX, rectMaxX) &&
        Math.max(circleMinY, rectMinY) <= Math.min(circleMaxY, rectMaxY)
    );
};
